## 思路

模拟。

工资按月累加，最后统一算税；劳务报酬每次单独算。

怎么算超额累进税率？从最高档往下扣，比如应税额 $x$，先看超过 $100000$ 的部分交 $45\%$，然后把 $x$ 截到 $100000$，再看超过 $80000$ 的部分交 $40\%$。以此类推。

唯一的坑点是同一个人同一个月可能有多条工资记录，要加起来算。开个二维数组存一下就好。

## Code
```cpp line-numbers
#include<bits/stdc++.h>
using namespace std;
#define int long long
int n,pay[50005][13];
string op;
double ans;
signed main() {
	ios::sync_with_stdio(false),cin.tie(0),cout.tie(0);
	cin>>n;
	while (cin>>op && op!="#") {
		int id;
		string d;
		double amt;
		cin>>id>>d>>amt;
		if (op=="PAY") {
			int m=stoi(d.substr(0,d.find('/')));
			pay[id][m]+=amt;
		}
		else if (op=="INCOME") {
			double x=(amt<=4000 ? max(0.0,amt-800.0) : amt*0.8);
			if (x>50000) ans+=(x-50000)*0.40,x=50000;
			if (x>20000) ans+=(x-20000)*0.30,x=20000;
			if (x>0) ans+=x*0.20;
		}
	}
	for (int i=1;i<=n;i++)
		for (int j=1;j<=12;j++) {
			if (pay[i][j]==0) continue;
			double x=max(0.0,pay[i][j]-800.0);
			if (x>100000) ans+=(x-100000)*0.45,x=100000;
			if (x>80000) ans+=(x-80000)*0.40,x=80000;
			if (x>60000) ans+=(x-60000)*0.35,x=60000;
			if (x>40000) ans+=(x-40000)*0.30,x=40000;
			if (x>20000) ans+=(x-20000)*0.25,x=20000;
			if (x>5000) ans+=(x-5000)*0.20,x=5000;
			if (x>2000) ans+=(x-2000)*0.15,x=2000;
			if (x>500) ans+=(x-500)*0.10,x=500;
			if (x>0) ans+=x*0.05;
		}
	cout<<fixed<<setprecision(2)<<ans;
	return 0;
}
```
