小测 T3。

## 思路

我们注意到这个重量具有单调性，所以我们可以直接把边权和重量离线下来然后从小到大排序。

然后我们就可以直接双指针。对于每一个询问 $(u,x)$，如果当前没加入的边满足 $w \le x$，那我们就可以直接用并查集将这条边的两个端点合并起来，然后累加集合大小即可。

## Code

```cpp
#include<bits/stdc++.h>
using namespace std;
#define int long long
const int maxn=1e5+5;
struct edge {
    int u,v,w;
}e[maxn];
struct node {
    int u,x,id;
}q[maxn];
bool cmp1(edge x,edge y) {
    return x.w<y.w;
}
bool cmp2(node x,node y) {
    return x.x<y.x;
}
int n,m,Q,fa[maxn],sz[maxn],ans[maxn];
int find(int x) {
    return fa[x]==x ? x : fa[x]=find(fa[x]);
}
void merge(int u,int v) {
    int fu=find(u),fv=find(v);
    if (fu==fv) return;
    fa[fu]=fv;
    sz[fv]+=sz[fu];
}
signed main() {
    ios::sync_with_stdio(false),cin.tie(0),cout.tie(0);
    cin>>n>>m>>Q;
    for (int i=1;i<=m;i++)
        cin>>e[i].u>>e[i].v>>e[i].w;
    for (int i=1;i<=Q;i++)
        cin>>q[i].u>>q[i].x,q[i].id=i;
    sort(e+1,e+1+m,cmp1);
    sort(q+1,q+1+Q,cmp2);
    for (int i=1;i<=n;i++)
        fa[i]=i,sz[i]=1;
    int cur=1;
    for (int i=1;i<=Q;i++) {
        while (cur<=m && e[cur].w<=q[i].x)
            merge(e[cur].u,e[cur].v),cur++;
        ans[q[i].id]=sz[find(q[i].u)];
    }
    for (int i=1;i<=Q;i++)
        cout<<ans[i]<<'\n';
    return 0;
}
```
