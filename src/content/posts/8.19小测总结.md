## T1 升序序列
贪心 + 数学。

我们先把 $b$ 都变为相同，然后比较 $a$，如果前一个数的 $a$ 更大，那么 $b$ 再加 $1$，即再乘 $2$。只需要遍历序列，对每个数进行以上操作即可。

可以将 $a_i$ 设为 $\log_2 a_i$，这样 $\times 2$ 就会变成 $+1$ 。

当然，此时提交会 WA 一些点，因为 double 的精度问题。可以先减去一个较小的数，然后再向上取整即可。

```cpp line-numbers
#include<bits/stdc++.h>
using namespace std;
#define int long long
#define db double
const db eps=1e-6; // 精度控制
int n,ans;
db a,b; // a: 前一个数调整后的 log2 值, b: 当前数的 log2 值
signed main() {
	ios::sync_with_stdio(false),cin.tie(0),cout.tie(0);
	cin>>n;
	while (n--) {
		int x;
		cin>>x;
		b=log2(x);
		if (a>b) {
			int x=ceil(a-b-eps);
			ans+=x
			b+=x;
		}
		a=b;
	}
	cout<<ans;
	return 0;
}
```

## T2 最大与求和
图论 + 数学推导。

考虑到联合的两个节点距离为 $2$，所以必定有一个中转点。所以，我们可以枚举每一个中转点。

假设每个中转点周围有两个点，权值分别为 $a,b$，则联合权值为 $2ab=(a+b)^2-(a^2+b^2)$。

若有三个点，权值分别为 $a,b,c$，则联合权值为 $2ab+2bc+2ac=(a+b+c)^2-(a^2+b^2+c^2)$。

所以，以某个节点为中转点的联合权值之和等于**权值和的平方减去权值的平方和**。

为了找到最大的联合权值，只需找到周围最大的两个权值，相乘判断即可。

```cpp line-numbers
#include<bits/stdc++.h>
using namespace std;
#define int long long
const int mod=10007;
int n,ans1,ans2,w[200005];
vector<int> e[200005];
signed main() {
	ios::sync_with_stdio(false),cin.tie(0),cout.tie(0);
	cin>>n;
	for (int i=1,u,v;i<n;i++)
		cin>>u>>v,e[u].push_back(v),e[v].push_back(u);
	for (int i=1;i<=n;i++)
		cin>>w[i];
	for (int i=1;i<=n;i++) {
		int max1=0,max2=0; //最大的两个权值
    int s1=0,s2=0; //t1 代表和的平方，t2 代表平方和
		for (auto ed:e[i]) {
			if (w[ed]>max1) max2=max1,max1=w[ed];
			else if (w[ed]>max2) max2=w[ed];
			s1=(s1+w[ed])%mod;
			s2=(s2+w[ed]*w[ed])%mod;
		}
		s1=s1*s1%mod;
		ans1=max(ans1,max1*max2);
		ans2=(ans2+s1+mod-s2)%mod
	}
	cout<<ans1<<" "<<ans2;
	return 0;
}
```

### 反思
注意取模细节，$s1^2 - s2$ 可能为负，要加 $mod$ 再取模。

## T3 子串
整个字符串排序后是唯一的，记为 $T$。

查询 $[l,r]$ 是否是 $T$ 的子串，等价于：
  1. $[l,r]$ 本身是**非降序**的。
  2. 区间内字符频次与 $T$ 中对应连续段一致。

所以可以用一个线段树维护区间最小值、最大值、左右端点字符、以及是否为非降序。

再用一个树状数组维护每种字符的出现次数。

查询时先判断是否非降序，再检查字符频次是否“连续”：即从最小字符到最大字符之间的每种字符数量必须与全局排序后该范围应有数量相等即可。

```cpp line-numbers
#include<bits/stdc++.h>
using namespace std;
#define int long long
#define lc 2*p
#define rc 2*p+1
const int maxn=1e5+5;
struct node {
	char minn,maxl,le,ri; // 区间最小字符、最大字符、左端字符、右端字符
	bool flag;            // 该区间是否是非降序的
}tree[4*maxn];
int n,q,cnt[30],c[30][maxn]; // cnt: 每种字符总数, c: 树状数组（每种字符的前缀和）
char s[maxn];
int lowbit(int x) {
	return x&(-x);
}
void add(int id,int x,int v) {
	for (;x<=n;x+=lowbit(x))
		c[id][x]+=v;
}
int sum(int id,int x){
	int res=0;
	for (;x>0;x-=lowbit(x))
		res+=c[id][x];
	return res;
}
node pushup(node a,node b) {
	node res;
	res.minn=min(a.minn,b.minn);
	res.maxl=max(a.maxl,b.maxl);
	res.le=a.le;
	res.ri=b.ri;
	res.flag=a.flag && b.flag && (a.ri<=b.le);
	return res;
}
void build(int p,int l,int r) {
	if (l==r) {
		char c=s[l];
		tree[p].minn=c,tree[p].maxl=c;
		tree[p].le=c,tree[p].ri=c;
		tree[p].flag=true;
		return;
	}
	int mid=(l+r)>>1;
	build(lc,l,mid);
	build(rc,mid+1,r);
	tree[p]=pushup(tree[lc],tree[rc]);
}
void update(int p,int l,int r,int idx,char c) {
	if (l==r) {
		tree[p].minn=c,tree[p].maxl=c;
		tree[p].le=c,tree[p].ri=c;
		tree[p].flag=true;
		return;
	}
	int mid=(l+r)/2;
	if (idx<=mid) update(lc,l,mid,idx,c);
	else update(rc,mid+1,r,idx,c);
	tree[p]=pushup(tree[lc],tree[rc]);
}
node query(int p,int s,int t,int l,int r) {
	if (l<=s && t<=r) return tree[p];
	int mid=(s+t)/2;
	if (r<=mid) return query(lc,s,mid,l,r);
	if (l>mid) return query(rc,mid+1,t,l,r);
	node a=query(lc,s,mid,l,r),b=query(rc,mid+1,t,l,r);
	return pushup(a,b);
}
signed main() {
	ios::sync_with_stdio(false),cin.tie(0),cout.tie(0);
	cin>>n>>(s+1)>>q;
	for (int i=1;i<=n;i++) {
		int x=s[i]-'a';
		cnt[x]++,add(x,i,1)
	}
	build(1,1,n);
	while (q--) {
		int op;
		cin>>op;
		if (op==1) {
			int x;
			char c;
			cin>>x>>c;
			if (s[x]==c) continue;
			int a=s[x]-'a',b=c-'a';
			add(a,x,-1),add(b,x,1)
			cnt[a]--,cnt[b]++;
			s[x]=c;
			update(1,1,n,x,c);
		}
		else {
			int l,r;
			cin>>l>>r;
			node t=query(1,1,n,l,r);
			if (!t.flag) {cout<<"No\n"; continue;} // 不是非降序，直接输出 No
			int a=t.minn-'a',b=t.maxl-'a';
			if (a==b) {cout<<"Yes\n"; continue;} // 只有一种字符，必为子串
			bool flag=true;
			// 检查从 a+1 到 b-1 的字符是否都“连续”出现
			for (int k=a+1;k<=b-1;k++){
				int s=sum(k,r)-sum(k,l-1); // [l,r] 中字符 'a'+k 的数量
				if (s!=cnt[k]) {flag=false; break;}
			}
			cout<<(flag ? "Yes\n" : "No\n");
		}
	}
	return 0;
}
```

## T4 三色划分
dp + 容斥。

构成三角形条件为任意两边之和大于第三边，即：
$$
R+G>B,\quad G+B>R,\quad B+R>G
$$

相当于 $最大边 < 总和 \div 2$。

总方案数为 $3^N$。考虑容斥：减去**不能构成三角形**的方案数。

$不能构成三角形 = 存在一边 \le 其余两边之和 = 存在一边 \le 总和的一半$。

枚举哪一边“过大”，用 DP 求出组和为 $j$ 的方案数。

但直接枚举会重复，但由于 $a_i \ge 1$ 且 $S \ge 3$，**不可能有两组同时** $\le S \div 2$。因此三类“过大”互斥，可以直接相加。

定义 $s_j$ 为用所有数，将某些数分给红组，红组和为 $j$ 的方案数（其余数可任意分给绿蓝）。

那么动态转移方程为：
$$
f_j = f_j \cdot 2 + s_{j-a_i}
$$
当前数不给红：$方案 \times 2$；给红：从 $j-a_i$ 转移）

$d_j$ 表示选出若干数和为 $j$ 的方案数（即红组恰好和为 $j$，绿蓝为空）。

```cpp line-numbers
#include<bits/stdc++.h>
using namespace std;
#define int long long
const int mod=998244353;
int n,sum,ans=1,a[305],s[90005],d[90005];
signed main() {
	ios::sync_with_stdio(false),cin.tie(0),cout.tie(0);
	cin>>n;
	for (int i=1;i<=n;i++)
		cin>>a[i];
	s[0]=3,d[0]=3;
	for (int i=1;i<=n;i++) {
		ans*=3,ans%=mod;
		sum+=a[i];
		for (int j=sum;j>=0;j--)
			s[j]=(s[j]*2%mod+s[max(0ll,j-a[i])])%mod;
		for (int j=sum;j>=a[i];j--)
			d[j]=(d[j]+d[j-a[i]])%mod;
	}
	int sum1=s[(sum+1)/2];
	if (sum%2==0) sum1=(sum1+mod-d[sum/2])%mod;
	ans=(ans+mod-sum1)%mod;
	cout<<ans%mod;
	return 0;
}
```
